```csharp
public Task DeleteAsync(int index)
{
    var provider = GetSqliteProvider()
        ?? throw new InvalidOperationException("SqliteConfigurationProvider not found");

    var cookieList = configurationRoot.GetSection("BiliBiliCookies").Get<List<string>>() ?? [];
    var newCount = cookieList.Count - 1;

    // Re-key all higher indices down by 1
    var rekeyDict = new Dictionary<string, string>();
    for (int i = index + 1; i < cookieList.Count; i++)
    {
        rekeyDict[$"BiliBiliCookies__{i - 1}"] = cookieList[i];
    }

    if (rekeyDict.Count > 0)
        provider.BatchSet(rekeyDict);

    // Delete the old last key (it's either the deleted one or has been re-keyed)
    provider.Set($"BiliBiliCookies__{newCount}", null);
    ReloadConfiguration();
    return Task.CompletedTask;
}
```

Note: `SqliteConfigurationProvider.Set()` with null value — the `BatchSet` method handles `DBNull.Value` for null, but `Set()` uses the value directly. Use `string.Empty` if null doesn't work, or use raw SQL DELETE for the last key. The executor should verify `Set(key, null)` behavior and adjust if needed.

```csharp
@code {
    [Inject] IDialogService DialogService { get; set; } = default!;

    private bool _loading = true;
    private List<BiliAccountDto> _accounts = [];

    protected override async Task OnInitializedAsync()
    {
        await LoadAccounts();
    }

    private async Task LoadAccounts()
    {
        _loading = true;
        _accounts = await Workflow.GetAllAccountsAsync();
        _loading = false;
    }

    private async Task OpenAddDialog()
    {
        var dialog = await DialogService.ShowAsync<AddAccountDialog>("Add Account");
        var result = await dialog.Result;
        if (!result.Canceled && result.Data is string cookieStr)
        {
            await Workflow.AddAsync(cookieStr);
            await LoadAccounts();
        }
    }

    private async Task OpenEditDialog(BiliAccountDto account)
    {
        var parameters = new DialogParameters { ["CookieStr"] = account.CookieStr };
        var dialog = await DialogService.ShowAsync<EditAccountDialog>("Edit Account", parameters);
        var result = await dialog.Result;
        if (!result.Canceled && result.Data is string newCookieStr)
        {
            await Workflow.UpdateAsync(account.Index, newCookieStr);
            await LoadAccounts();
        }
    }

    private async Task OpenDeleteDialog(BiliAccountDto account)
    {
        var parameters = new DialogParameters { ["UserId"] = account.UserId };
        var dialog = await DialogService.ShowAsync<DeleteAccountDialog>("Delete Account", parameters);
        var result = await dialog.Result;
        if (!result.Canceled)
        {
            await Workflow.DeleteAsync(account.Index);
            await LoadAccounts();
        }
    }
}
```
